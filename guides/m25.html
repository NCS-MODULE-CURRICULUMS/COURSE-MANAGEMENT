절차 2 — 구 시스템을 만든다

CREATE DATABASE oldsys DEFAULT CHARACTER SET utf8mb4; USE oldsys; CREATE TABLE 예약대장 ( 일련번호 VARCHAR(20), 회원명 VARCHAR(50), 연락처 VARCHAR(30), 시설명 VARCHAR(50), 이용일자 VARCHAR(20), 인원 VARCHAR(10), 금액 VARCHAR(20), 비고 VARCHAR(100) );

절차 4 — 신 시스템을 만든다

CREATE DATABASE newsys DEFAULT CHARACTER SET utf8mb4; USE newsys; CREATE TABLE member ( member_id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(20) NOT NULL, phone VARCHAR(20), UNIQUE KEY uk_member (name, phone) ); CREATE TABLE facility ( facility_id INT PRIMARY KEY, name VARCHAR(50) NOT NULL UNIQUE ); CREATE TABLE reservation ( reservation_id INT AUTO_INCREMENT PRIMARY KEY, old_no VARCHAR(20) NOT NULL UNIQUE, member_id INT NOT NULL, facility_id INT NOT NULL, use_date DATE NOT NULL, headcount INT NOT NULL, fee INT NOT NULL, FOREIGN KEY (member_id) REFERENCES member(member_id), FOREIGN KEY (facility_id) REFERENCES facility(facility_id) ); INSERT INTO facility VALUES (1,'중앙체육관'),(2,'수성풋살장'),(3,'달성운동장');

절차 6 — 몇 건이 들어갔는지 센다

+--------------+--------+--------+ | 구시스템 | 회원 | 예약 | +--------------+--------+--------+ | 9 | 8 | 0 | +--------------+--------+--------+

예약이 0건입니다. 회원 8명은 들어갔는데 예약은 한 건도 안 들어갔습니다. 날짜 구분자가 슬래시인 행 하나 때문에 전체가 멈췄습니다. 나머지 여덟 건은 멀쩡했는데도 그렇습니다. 자료를 옮기기 전에 옮길 수 없는 행을 먼저 골라내는 일이 이 교과에서 가장 중요한 절차인 까닭입니다.
절차 7 — 회원이 8명인 까닭을 확인한다

구 시스템의 행은 9개인데 회원은 8명입니다. 마지막 행이 첫 행과 이름·연락처가 같은 중복이라 UNIQUE KEY 가 걸러 냈습니다. INSERT IGNORE 를 썼기 때문에 오류 없이 넘어간 것입니다.

① facility 시설 — 미리 넣어 둡니다 ② member 회원 — 예약이 가리키므로 먼저 ③ reservation 예약 — 마지막

외래키가 걸린 표를 나중에 옮깁니다. 예약을 먼저 옮기면 가리킬 회원이 없어서 막힙니다. 응용SW 기초 기술 활용 교과에서 본 1452 오류입니다.

전환 SQL

-- 회원 INSERT IGNORE INTO member (name, phone) SELECT DISTINCT 회원명, 연락처 FROM oldsys.예약대장_정비; -- 예약 INSERT INTO reservation (old_no, member_id, facility_id, use_date, headcount, fee) SELECT o.일련번호, m.member_id, f.facility_id, STR_TO_DATE(o.이용일자,'%Y-%m-%d'), CAST(o.인원 AS UNSIGNED), CAST(o.금액 AS UNSIGNED) FROM oldsys.예약대장_정비 o JOIN member m ON m.name = o.회원명 AND (m.phone <=> o.연락처) JOIN facility f ON f.name = o.시설명;

<=> 를 쓴 것에 주의하십시오. = 는 NULL 끼리 비교하면 참이 안 됩니다. 연락처가 없는 회원이 있으므로 NULL 도 같다고 보는 연산자가 필요합니다.
